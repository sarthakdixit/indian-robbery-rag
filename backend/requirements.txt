# Runtime dependencies for the backend container.
#
# Pinned with ~= (compatible release) to get patch updates without
# minor-version breakage. Verify a fresh `pip install -r requirements.txt`
# inside the container after any update — pip resolution differs
# slightly from local venvs.
#
# Azure SDKs (cosmos, identity, monitor-opentelemetry) are included
# even though they're only loaded in cloud mode. Their cost in the
# image is acceptable (~30MB) and including them keeps the local-vs-
# cloud swap a one-flag change.

# --- HTTP / ASGI ----------------------------------------------------
fastapi ~= 0.139.0
uvicorn[standard] ~= 0.50.0

# --- Validation / settings ----------------------------------------
pydantic ~= 2.13.0
pydantic-settings ~= 2.14.0

# --- DI -----------------------------------------------------------
dependency-injector ~= 4.49.0

# --- Storage / cache ----------------------------------------------
aiosqlite ~= 0.22.0

# --- Outbound HTTP (Turnstile verification, future Cloudflare ----
# API calls). Replaces `requests` (sync) which AGENT.md §7.1 forbids.
httpx ~= 0.28.0

# --- Logging ------------------------------------------------------
# Pinned even though we still emit through stdlib logging — the
# structlog migration is deferred tech debt. Including it now avoids
# needing a `requirements.txt` bump when the migration lands.
structlog ~= 26.1.0

# --- Retrieval ----------------------------------------------------
# ChromaDB ships prebuilt wheels for linux/amd64 (the slim base) so
# no compile toolchain is needed in the runtime stage.
chromadb ~= 1.5.0
bm25s ~= 0.3.0

# --- Gemini SDK ---------------------------------------------------
# Use the newer `google-genai` (NOT `google-generativeai`). AGENT.md
# §17.3 mentions the legacy one; the actual code under backend/
# imports `from google import genai`.
google-genai ~= 2.8.0

# --- Azure SDKs (cloud mode only) ---------------------------------
# Lazy-imported by the cloud adapters; local mode never loads them
# but they're bundled to keep the cloud swap a config flip rather
# than an image rebuild.
azure-cosmos ~= 4.15.0
# aiohttp is REQUIRED for azure.cosmos.aio (the async client) but
# azure-cosmos doesn't declare it as a hard dependency — it's an
# optional `[aio]` extra that may not be available on every version.
# We list it directly so the async transport works regardless of
# which azure-cosmos point version pip resolves to.
# See: https://github.com/Azure/azure-sdk-for-python/issues/41737
aiohttp ~= 3.10.11
# azure-identity is required by azure-monitor-opentelemetry (for
# managed identity auth in App Insights ingestion). We don't import
# it directly from our code.
azure-identity ~= 1.20.0
azure-monitor-opentelemetry ~= 1.8.0
